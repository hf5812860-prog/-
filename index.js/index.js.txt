const KEY_FILE = path.join(__dirname, 'key.txt');

function generateKey() {
    // 32 حرف عشوائي (آمن)
    return 'rk_' + crypto.randomBytes(24).toString('hex');
}

function loadOrCreateKey() {
    // 1. لو Environment Variable موجود → استخدمه
    if (process.env.API_KEY) {
        console.log('🔑 Using API_KEY from environment');
        return process.env.API_KEY;
    }

    // 2. لو ملف key.txt موجود → اقرأه
    if (fs.existsSync(KEY_FILE)) {
        const saved = fs.readFileSync(KEY_FILE, 'utf8').trim();
        if (saved) {
            console.log('🔑 Loaded key from key.txt');
            return saved;
        }
    }

    // 3. ولّد مفتاح جديد
    const newKey = generateKey();
    try {
        fs.writeFileSync(KEY_FILE, newKey, 'utf8');
        console.log('🔑 Generated new key → saved to key.txt');
    } catch (e) {
        console.warn('⚠️ Could not save key.txt (read-only filesystem?)');
        console.warn('   Key will reset on restart. Set API_KEY env var to persist.');
    }

    return newKey;
}

// 🔑 المفتاح مُمرر من السيرفر بأمان
const API_KEY = "${API_KEY}";

function timeAgo(ms) {
    const s = Math.floor((Date.now() - ms) / 1000);
    if (s < 60)   return s + ' ثانية';
    if (s < 3600) return Math.floor(s / 60) + ' دقيقة';
    if (s < 86400) return Math.floor(s / 3600) + ' ساعة';
    return Math.floor(s / 86400) + ' يوم';
}

function formatUptime(ms) {
    const s = Math.floor(ms / 1000);
    const h = Math.floor(s / 3600);
    const m = Math.floor((s % 3600) / 60);
    if (h > 0) return h + 'h ' + m + 'm';
    if (m > 0) return m + 'm ' + (s % 60) + 's';
    return s + 's';
}

function escapeHtml(str) {
    return String(str || '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
}

// 🔑 نسخ المفتاح
function copyKey() {
    const keyText = document.getElementById('api-key').textContent.trim();
    const btn = document.getElementById('copy-btn');
    const icon = document.getElementById('copy-icon');
    const label = document.getElementById('copy-text');

    navigator.clipboard.writeText(keyText).then(() => {
        btn.classList.add('copied');
        icon.textContent = '✅';
        label.textContent = 'تم النسخ!';
        showToast('✅ تم نسخ المفتاح بنجاح');

        setTimeout(() => {
            btn.classList.remove('copied');
            icon.textContent = '📋';
            label.textContent = 'نسخ المفتاح';
        }, 2000);
    }).catch(err => {
        // fallback
        const textarea = document.createElement('textarea');
        textarea.value = keyText;
        document.body.appendChild(textarea);
        textarea.select();
        document.execCommand('copy');
        document.body.removeChild(textarea);
        showToast('✅ تم النسخ');
    });
}

function showToast(msg) {
    const t = document.getElementById('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(() => t.classList.remove('show'), 2000);
}

async function fetchStats() {
    try {
        const res = await fetch('/dashboard/stats', {
            headers: { 'x-api-key': API_KEY }
        });
        if (!res.ok) return;
        const data = await res.json();
        document.getElementById('stat-users').textContent  = data.users;
        document.getElementById('stat-trades').textContent = data.trades;
        document.getElementById('stat-total').textContent  = data.totalTrades;
        document.getElementById('stat-uptime').textContent = formatUptime(data.uptime);
        document.getElementById('uptime').textContent      = formatUptime(data.uptime);
    } catch (e) { console.error(e); }
}

async function fetchTrades() {
    try {
        const res = await fetch('/api/trades/all', {
            headers: { 'x-api-key': API_KEY }
        });
        if (!res.ok) return;
        const trades = await res.json();

        document.getElementById('trades-count').textContent = trades.length + ' عرض';

        const list = document.getElementById('trades-list');
        if (trades.length === 0) {
            list.innerHTML = '<div class="empty"><div class="empty-icon">📭</div><div class="empty-text">لا توجد عروض حالياً</div></div>';
            return;
        }

        list.innerHTML = trades.map(t => {
            const avatar = 'https://www.roblox.com/headshot-thumbnail/image?userId=' + (t.fromId || 1) + '&width=150&height=150&format=png';
            const gives = (t.myItems || []).join(', ') || '—';
            const wants = (t.theirItems || []).join(', ') || 'أي عرض';
            const note  = t.note ? '<div class="trade-row"><span class="label">💬</span><span class="trade-items">' + escapeHtml(t.note) + '</span></div>' : '';

            return \`
            <div class="trade-card">
                <img class="trade-avatar" src="\${avatar}" onerror="this.style.display='none'">
                <div class="trade-info">
                    <div class="trade-name">👤 \${escapeHtml(t.fromName)}</div>
                    <div class="trade-row">
                        <span class="label">🎁 يعطي:</span>
                        <span class="trade-items">\${escapeHtml(gives)}</span>
                    </div>
                    <div class="trade-row">
                        <span class="label">🎯 يبي:</span>
                        <span class="trade-items">\${escapeHtml(wants)}</span>
                    </div>
                    \${note}
                    <div class="trade-time">⏱️ منذ \${timeAgo(t.createdAt)}</div>
                </div>
            </div>\`;
        }).join('');
    } catch (e) { console.error(e); }
}

async function fetchLogs() {
    try {
        const res = await fetch('/dashboard/logs', {
            headers: { 'x-api-key': API_KEY }
        });
        if (!res.ok) return;
        const logs = await res.json();

        const list = document.getElementById('logs-list');
        if (logs.length === 0) {
            list.innerHTML = '<div class="empty"><div class="empty-icon">📋</div><div class="empty-text">لا توجد أحداث</div></div>';
            return;
        }

        const icons = {
            register: '👤',
            create:   '📤',
            delete:   '🗑️',
            accept:   '✅',
            reject:   '❌',
        };

        list.innerHTML = logs.map(l => \`
            <div class="log-item \${l.type}">
                <div class="log-icon">\${icons[l.type] || '📌'}</div>
                <div class="log-content">
                    <div class="log-msg">\${escapeHtml(l.message)}</div>
                    <div class="log-time">منذ \${timeAgo(l.time)}</div>
                </div>
            </div>
        \`).join('');
    } catch (e) { console.error(e); }
}

app.post('/api/register', auth, (req, res) => {
    const { robloxId, username, jobId } = req.body;
    if (!robloxId) return res.status(400).json({ error: 'Missing robloxId' });

    users[robloxId] = { robloxId, username, jobId, lastSeen: Date.now() };
    addLog('register', `${username} سجّل دخول (ID: ${robloxId})`);
    res.json({ success: true });
});

app.post('/api/trade/create', auth, (req, res) => {
    const { fromId, fromName, myItems, theirItems, note, jobId } = req.body;
    if (!fromId) return res.status(400).json({ error: 'Missing fromId' });

    const tradeId = 'tr_' + Date.now().toString(36) + Math.random().toString(36).slice(2, 6);

    trades[tradeId] = {
        id: tradeId, fromId, fromName,
        myItems: Array.isArray(myItems) ? myItems : [],
        theirItems: Array.isArray(theirItems) ? theirItems : [],
        note: note || "",
        jobId: jobId || "",
        status: 'pending',
        createdAt: Date.now()
    };

    totalTrades++;
    addLog('create', `${fromName} نشر عرضاً: يعطي [${(myItems || []).join(', ')}] / يبي [${(theirItems || []).join(', ')}]`);

    res.json({ success: true, tradeId });
});

app.get('/api/trades/all', auth, (req, res) => {
    const cutoff = Date.now() - 24 * 60 * 60 * 1000;
    const list = Object.values(trades).filter(t =>
        t.status === 'pending' && t.createdAt > cutoff
    );
    list.sort((a, b) => b.createdAt - a.createdAt);
    res.json(list);
});

app.post('/api/trade/:id/delete', auth, (req, res) => {
    const trade = trades[req.params.id];
    if (!trade) return res.status(404).json({ error: 'Not found' });
    if (trade.fromId != req.body.fromId) return res.status(403).json({ error: 'Not yours' });

    addLog('delete', `${trade.fromName} حذف عرضه`);
    delete trades[req.params.id];
    res.json({ success: true });
});

app.post('/api/trade/:id/respond', auth, (req, res) => {
    const { accept, responderId } = req.body;
    const trade = trades[req.params.id];

    if (!trade) return res.status(404).json({ error: 'Not found' });
    if (trade.fromId == responderId) return res.status(403).json({ error: 'Own trade' });

    trade.status = accept ? 'accepted' : 'rejected';
    trade.respondedAt = Date.now();
    trade.responderId = responderId;

    addLog(accept ? 'accept' : 'reject',
           `${users[responderId]?.username || responderId} ${accept ? 'قبل' : 'رفض'} عرض ${trade.fromName}`);

    res.json({ success: true, trade });
});
